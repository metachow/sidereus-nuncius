% Front and back cover of the edition.
% In the manner of an early seventeenth-century title page: paper ground,
% black and Venetian red, spaced capitals, an engraving printed in ink.
\definecolor{coverpaper}{RGB}{243,236,220}
\definecolor{coverink}{RGB}{38,30,24}
\definecolor{coverred}{RGB}{139,42,31}

% paper ground and a double rule (heavy red outside, thin ink inside)
\newcommand{\coverframe}{%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[coverpaper] (current page.south west) rectangle (current page.north east);
    \draw[coverred, line width=2.2pt]
      ([shift={(11mm,11mm)}]current page.south west) rectangle ([shift={(-11mm,-11mm)}]current page.north east);
    \draw[coverink, line width=0.5pt]
      ([shift={(14mm,14mm)}]current page.south west) rectangle ([shift={(-14mm,-14mm)}]current page.north east);
  \end{tikzpicture}}
